import hashlib
import json
import os
import pathlib
import subprocess
import sys

root = pathlib.Path(os.environ.get('QI_SOURCE_ROOT', '/Users/songmin/Documents/code-new/quanta-index'))
seed = json.loads(pathlib.Path(os.environ.get('QI_SOURCE_SEED', str(root / 'docs/plans/sep-27-code-search-remediation/handoffs/l1-l5-proof/workspace-source-contract-post.json'))).read_text())
files = set(seed['files'])
listed = subprocess.check_output(['git', 'ls-files', '--cached', '--others', '--exclude-standard', '-z'], cwd=root).decode().split('\0')
for name in listed:
    if name and (name.endswith(('.rs', '.toml', '.py', '.sh')) or name in ('Cargo.lock', 'Justfile')):
        files.add(name)

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

local = {name: sha(root / name) for name in sorted(files) if (root / name).is_file()}
external_root = pathlib.Path(seed['external_path'])
external = {name: sha(external_root / name) for name in sorted(seed['external_files']) if (external_root / name).is_file()}
head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip()
external_head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=external_root, text=True).strip()
dirty = subprocess.check_output(['git', 'status', '--short'], cwd=root, text=True).splitlines()
identity = {'files': local, 'external_files': external, 'head': head, 'external_head': external_head}
digest = hashlib.sha256(json.dumps(identity, sort_keys=True, separators=(',', ':')).encode()).hexdigest()
manifest = {'identity': identity, 'source_digest': digest, 'dirty': dirty, 'external_path': str(external_root), 'hash_algorithm': 'SHA-256 over canonical JSON identity; per-file SHA-256 raw bytes'}
path = root / sys.argv[1]
path.write_text(json.dumps(manifest, indent=2, sort_keys=True) + '\n')
print(f'{path}: files={len(local)} external={len(external)} digest={digest} head={head} dirty={len(dirty)}')
